Görüntü Sınıflandırma Modeli Yapımı

📚 G2 · Görüntü Serisi⏱ ~40 dakika🖥 İnteraktif Demo
Bu derste öğrenecekleriniz: MNIST veri setiyle uçtan uca bir CNN sınıflandırıcı eğitmek: veri yükleme, model tanımı, eğitim döngüsü, karışıklık matrisi ile hata analizi ve hatalı tahminleri inceleme. Bu dersin sonunda kendi modelinizi Colab'da eğitebileceksiniz.

End-to-End Proje: Rakam Tanıma (MNIST)

MNIST: 28×28 gri el yazısı rakam (0-9) veri seti — 60.000 eğitim, 10.000 test örneği. Yapay zekanın "Merhaba Dünya"sıdır. Küçük ama gerçek derin öğrenme problemidir.

Adım 1: Veri Yükleme

import torch
from torchvision import datasets, transforms
from torch.utils.data import DataLoader

transform = transforms.Compose([
    transforms.ToTensor(),               # [0,255] → [0,1] float tensor
    transforms.Normalize((0.1307,), (0.3081,))  # MNIST istatistikleriyle normalize
])

train_data = datasets.MNIST("./data", train=True, download=True, transform=transform)
test_data  = datasets.MNIST("./data", train=False, download=True, transform=transform)

train_loader = DataLoader(train_data, batch_size=64, shuffle=True)
test_loader  = DataLoader(test_data, batch_size=64)

print(len(train_data))   # 60000
print(train_data[0][0].shape)  # torch.Size([1, 28, 28])

Adım 2: CNN Model Tanımı

import torch.nn as nn

class RakamSiniflandirici(nn.Module):
    def __init__(self):
        super().__init__()
        self.conv1 = nn.Conv2d(1, 32, 3, padding=1)   # 1→32 kanal, 3x3
        self.conv2 = nn.Conv2d(32, 64, 3, padding=1)  # 32→64 kanal
        self.pool = nn.MaxPool2d(2, 2)                 # 2x küçültme
        self.fc1 = nn.Linear(64 * 7 * 7, 128)         # 64×7×7 = 3136
        self.fc2 = nn.Linear(128, 10)                  # 10 rakam sınıfı
        self.dropout = nn.Dropout(0.25)
        self.relu = nn.ReLU()

    def forward(self, x):
        x = self.pool(self.relu(self.conv1(x)))  # 28→14
        x = self.pool(self.relu(self.conv2(x)))  # 14→7
        x = x.view(x.size(0), -1)                # Düzleştir: (B, 3136)
        x = self.dropout(self.relu(self.fc1(x)))
        return self.fc2(x)                       # Ham skorlar (logits)

model = RakamSiniflandirici()
print(f"Parametre: {sum(p.numel() for p in model.parameters()):,}")  # ~421K

Adım 3: Eğitim Döngüsü

device = "cuda" if torch.cuda.is_available() else "cpu"
model = model.to(device)
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
loss_fn = nn.CrossEntropyLoss()

for epoch in range(3):
    model.train()
    toplam_kayip = 0
    for X, y in train_loader:
        X, y = X.to(device), y.to(device)
        optimizer.zero_grad()
        kayip = loss_fn(model(X), y)
        kayip.backward()
        optimizer.step()
        toplam_kayip += kayip.item()
    print(f"Epoch {epoch+1}: kayıp={toplam_kayip/len(train_loader):.4f}")

Adım 4: Değerlendirme ve Karışıklık Matrisi

from sklearn.metrics import confusion_matrix
import numpy as np

model.eval()   # eval modu: dropout kapat
tum_tahminler, tum_gercekler = [], []

with torch.no_grad():
    for X, y in test_loader:
        tahmin = model(X.to(device)).argmax(dim=1).cpu()
        tum_tahminler.extend(tahmin.tolist())
        tum_gercekler.extend(y.tolist())

dogruluk = np.mean(np.array(tum_tahminler) == np.array(tum_gercekler))
print(f"Test doğruluğu: %{dogruluk*100:.1f}")   # ~%99 beklenor

conf = confusion_matrix(tum_gercekler, tum_tahminler)
# Hangi rakamlar karışıyor? Sıklıkla 4/9 ve 3/8
